 Excellente question ! Pour déployer votre chatbot Rasa de

  🏗️ Architecture de Production Recommandée

  1. Infrastructure Sécurisée

  Internet → Reverse Proxy (Nginx/Apache) → Rasa Server +
  Action Server
                  ↓ HTTPS                        ↓ HTTP
  interne
              Certificat SSL               Base de données        
  sécurisée

  2. Composants à Déployer

  Serveur Web (Nginx/Apache) :
  - Certificat SSL/TLS (Let's Encrypt gratuit)
  - Reverse proxy vers Rasa
  - Configuration CORS pour multi-sites
  - Rate limiting
  - Logs d'accès

  Rasa + Actions :
  - rasa run --enable-api --cors "*" --port 5005
  - rasa run actions --port 5055
  - Variables d'environnement pour secrets

  Base de Données :
  - SQL Server avec authentification sécurisée
  - Connexions chiffrées (TLS)
  - Utilisateur dédié avec permissions limitées
  - Sauvegarde automatique

> configuration Nginx


● Voici une configuration Nginx complète et sécurisée pour        
  votre chatbot Rasa :

  🔧 Configuration Nginx Complète

  1. Structure des Fichiers

  /etc/nginx/
  ├── nginx.conf                    # Configuration
  principale
  ├── sites-available/
  │   └── chatbot.fonctionpublique   # Configuration du site      
  ├── sites-enabled/
  │   └── chatbot.fonctionpublique   # Lien symbolique
  └── ssl/
      ├── chatbot.crt               # Certificat SSL
      └── chatbot.key               # Clé privée

  2. Configuration Principale (/etc/nginx/nginx.conf)

  user www-data;
  worker_processes auto;
  pid /run/nginx.pid;

  events {
      worker_connections 1024;
      use epoll;
      multi_accept on;
  }

  http {
      # Configuration de base
      sendfile on;
      tcp_nopush on;
      tcp_nodelay on;
      keepalive_timeout 65;
      types_hash_max_size 2048;
      client_max_body_size 10M;

      # MIME types
      include /etc/nginx/mime.types;
      default_type application/octet-stream;

      # Logs
      log_format main '$remote_addr - $remote_user
  [$time_local] "$request" '
                     '$status $body_bytes_sent
  "$http_referer" '
                     '"$http_user_agent"
  "$http_x_forwarded_for" '
                     'rt=$request_time
  uct="$upstream_connect_time" '
                     'uht="$upstream_header_time"
  urt="$upstream_response_time"';

      access_log /var/log/nginx/access.log main;
      error_log /var/log/nginx/error.log warn;

      # Gzip compression
      gzip on;
      gzip_vary on;
      gzip_min_length 1024;
      gzip_proxied any;
      gzip_comp_level 6;
      gzip_types
          text/plain
          text/css
          text/xml
          text/javascript
          application/json
          application/javascript
          application/xml+rss
          application/atom+xml;

      # Rate limiting zones
      limit_req_zone $binary_remote_addr zone=api:10m
  rate=10r/m;
      limit_req_zone $binary_remote_addr zone=chat:10m
  rate=30r/m;

      # Upstreams pour load balancing
      upstream rasa_server {
          server 127.0.0.1:5005 max_fails=3 fail_timeout=30s;     
          # Ajoutez d'autres serveurs Rasa si nécessaire
          # server 127.0.0.1:5006 max_fails=3
  fail_timeout=30s;
      }

      upstream rasa_actions {
          server 127.0.0.1:5055 max_fails=3 fail_timeout=30s;     
      }

      # Inclure les sites
      include /etc/nginx/sites-enabled/*;
  }

  3. Configuration du Site 
  (/etc/nginx/sites-available/chatbot.fonctionpublique)

  # Redirection HTTP vers HTTPS
  server {
      listen 80;
      listen [::]:80;
      server_name chatbot.fonctionpublique.gouv.sn;

      # Redirection permanente vers HTTPS
      return 301 https://$server_name$request_uri;
  }

  # Configuration HTTPS principale
  server {
      listen 443 ssl http2;
      listen [::]:443 ssl http2;
      server_name chatbot.fonctionpublique.gouv.sn;

      # ===================
      # CONFIGURATION SSL
      # ===================
      ssl_certificate /etc/nginx/ssl/chatbot.crt;
      ssl_private_key /etc/nginx/ssl/chatbot.key;

      # Configuration SSL moderne et sécurisée
      ssl_protocols TLSv1.2 TLSv1.3;
      ssl_ciphers
  ECDHE-RSA-AES256-GCM-SHA512:DHE-RSA-AES256-GCM-SHA512:ECDHE     
  -RSA-AES256-GCM-SHA384:DHE-RSA-AES256-GCM-SHA384;
      ssl_prefer_server_ciphers off;
      ssl_session_cache shared:SSL:10m;
      ssl_session_timeout 10m;
      ssl_session_tickets off;

      # HSTS (Strict Transport Security)
      add_header Strict-Transport-Security "max-age=63072000;     
   includeSubDomains; preload" always;

      # ===================
      # HEADERS DE SÉCURITÉ
      # ===================
      add_header X-Frame-Options DENY always;
      add_header X-XSS-Protection "1; mode=block" always;
      add_header X-Content-Type-Options nosniff always;
      add_header Referrer-Policy
  "strict-origin-when-cross-origin" always;
      add_header Content-Security-Policy "default-src 'self';     
   script-src 'self' 'unsafe-inline'; style-src 'self'
  'unsafe-inline'; img-src 'self' data:; connect-src 'self'       
  wss:;" always;

      # CORS Headers pour intégration multi-sites
      set $cors_origin "";
      set $cors_methods "GET, POST, OPTIONS";
      set $cors_headers
  "DNT,User-Agent,X-Requested-With,If-Modified-Since,Cache-Co     
  ntrol,Content-Type,Range,Authorization";

      # Domaines autorisés
      if ($http_origin ~*
  ^https?://(pgde\.gouv\.sn|e-carriere\.sec\.gouv\.sn|fonctio     
  npublique\.gouv\.sn|crce\.gouv\.sn)$) {
          set $cors_origin $http_origin;
      }

      add_header Access-Control-Allow-Origin $cors_origin
  always;
      add_header Access-Control-Allow-Methods $cors_methods       
  always;
      add_header Access-Control-Allow-Headers $cors_headers       
  always;
      add_header Access-Control-Allow-Credentials true
  always;

      # ===================
      # GESTION DES OPTIONS (PREFLIGHT)
      # ===================
      location ~ ^/(webhooks|conversations) {
          if ($request_method = 'OPTIONS') {
              add_header Access-Control-Allow-Origin
  $cors_origin always;
              add_header Access-Control-Allow-Methods
  $cors_methods always;
              add_header Access-Control-Allow-Headers
  $cors_headers always;
              add_header Access-Control-Max-Age 86400;
              add_header Content-Type 'text/plain;
  charset=utf-8';
              add_header Content-Length 0;
              return 204;
          }
      }

      # ===================
      # ENDPOINT CHATBOT PRINCIPAL
      # ===================
      location /webhooks/rest/webhook {
          # Rate limiting pour éviter le spam
          limit_req zone=chat burst=5 nodelay;

          # Proxy vers Rasa
          proxy_pass
  http://rasa_server/webhooks/rest/webhook;
          proxy_set_header Host $host;
          proxy_set_header X-Real-IP $remote_addr;
          proxy_set_header X-Forwarded-For
  $proxy_add_x_forwarded_for;
          proxy_set_header X-Forwarded-Proto $scheme;

          # Timeouts
          proxy_connect_timeout 10s;
          proxy_send_timeout 30s;
          proxy_read_timeout 30s;

          # Headers pour CORS
          add_header Access-Control-Allow-Origin $cors_origin     
   always;
          add_header Access-Control-Allow-Credentials true        
  always;
      }

      # ===================
      # API CONVERSATIONS (Optionnel)
      # ===================
      location /conversations {
          # Rate limiting plus strict pour l'API
          limit_req zone=api burst=2 nodelay;

          # Authentification basique (à adapter)
          auth_basic "API Chatbot";
          auth_basic_user_file /etc/nginx/.htpasswd;

          proxy_pass http://rasa_server/conversations;
          proxy_set_header Host $host;
          proxy_set_header X-Real-IP $remote_addr;
          proxy_set_header X-Forwarded-For
  $proxy_add_x_forwarded_for;
          proxy_set_header X-Forwarded-Proto $scheme;
      }

      # ===================
      # HEALTH CHECK
      # ===================
      location /health {
          access_log off;
          proxy_pass http://rasa_server/;
          proxy_set_header Host $host;

          # Headers personnalisés pour monitoring
          add_header X-Chatbot-Status "OK" always;
      }

      # ===================
      # STATIC ASSETS (Widget JS, CSS)
      # ===================
      location /static/ {
          alias /var/www/chatbot/static/;
          expires 1y;
          add_header Cache-Control "public, immutable";
          add_header Access-Control-Allow-Origin $cors_origin     
   always;
      }

      # ===================
      # GESTION DES ERREURS
      # ===================
      error_page 404 /404.html;
      error_page 500 502 503 504 /50x.html;

      location = /404.html {
          root /var/www/chatbot/errors;
          internal;
      }

      location = /50x.html {
          root /var/www/chatbot/errors;
          internal;
      }

      # ===================
      # SÉCURITÉ SUPPLÉMENTAIRE
      # ===================
      # Bloquer les bots malveillants
      if ($http_user_agent ~* (bot|crawler|spider|scraping))      
  {
          return 403;
      }

      # Bloquer les requêtes sans User-Agent
      if ($http_user_agent = "") {
          return 403;
      }

      # Limiter la taille des requêtes
      client_max_body_size 1M;

      # Désactiver les logs pour certains paths
      location = /favicon.ico {
          access_log off;
          log_not_found off;
      }

      location = /robots.txt {
          access_log off;
          log_not_found off;
          return 200 "User-agent: *\nDisallow: /\n";
      }
  }

  # ===================
  # CONFIGURATION MONITORING (Optionnel)
  # ===================
  server {
      listen 127.0.0.1:8080;
      server_name localhost;

      location /nginx_status {
          stub_status on;
          access_log off;
          allow 127.0.0.1;
          deny all;
      }
  }
