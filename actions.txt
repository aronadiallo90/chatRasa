import os
import pdfplumber
import ollama
import difflib
from typing import Text, Dict, Any
from rasa_sdk import Action, Tracker
from rasa_sdk.executor import CollectingDispatcher
from rasa_sdk.events import UserUtteranceReverted

# === Chargement du document PDF (Découpage en Sections) ===
DOCUMENT_PATH = "data/knowledge_base/faq.pdf"

def extract_text_from_pdf(pdf_path):
    """ Extraction du texte depuis un PDF, divisé en sections """
    sections = []
    with pdfplumber.open(pdf_path) as pdf:
        for page in pdf.pages:
            extracted_text = page.extract_text()
            if extracted_text:
                sections.append(extracted_text)  # On stocke chaque page comme une section
    return sections

DOCUMENT_SECTIONS = extract_text_from_pdf(DOCUMENT_PATH)

# Cache pour stocker les réponses aux questions récurrentes
response_cache = {}

class ActionRetrieveAnswer(Action):
    """ Action pour chercher une réponse en optimisant la requête LLaMA """

    def name(self) -> str:
        return "action_retrieve_answer"

    def run(self, dispatcher, tracker, domain):
        user_message = tracker.latest_message.get("text", "").lower()

        # 📌 1️⃣ Vérifier si la réponse est en cache
        if user_message in response_cache:
            dispatcher.utter_message(text=response_cache[user_message])
            return []

        # 📌 2️⃣ Vérifier si la question correspond à une réponse déjà définie
        predefined_answers = {
            "horaires": "Nos horaires sont du lundi au vendredi de 9h à 17h.",
            "créer un compte": "Rendez-vous sur notre site et cliquez sur 'Inscription'.",
            "connexion": "Utilisez vos identifiants pour vous connecter.",
            "lien tutoriel": "Voici le tutoriel : [Lien du tutoriel]",
            "postuler à la fonction publique": "Le processus d'inscription est détaillé sur notre site."
        }
        for key, answer in predefined_answers.items():
            if key in user_message:
                dispatcher.utter_message(text=answer)
                return []

        # 📌 3️⃣ Trouver la section la plus pertinente dans le document
        best_section = max(DOCUMENT_SECTIONS, key=lambda sec: difflib.SequenceMatcher(None, sec.lower(), user_message).ratio())

        # 📌 4️⃣ Générer une réponse avec Ollama en utilisant SEULEMENT la section trouvée
        prompt = f"""
        Tu es un assistant qui répond aux questions en utilisant les informations suivantes :
        ---
        {best_section}
        ---
        Question de l'utilisateur : {user_message}
        Réponds de manière claire et concise.
        """

        try:
            response = ollama.chat(
                model="llama3",
                messages=[{"role": "user", "content": prompt}]
            )
            bot_response = response["message"]["content"]
        
        except Exception:
            bot_response = "Désolé, je ne peux pas répondre pour le moment."

        # 📌 5️⃣ Ajouter la réponse en cache pour accélérer les requêtes similaires
        response_cache[user_message] = bot_response

        dispatcher.utter_message(text=bot_response)

        # Ajouter un bouton de retour
        dispatcher.utter_message(text="Que souhaitez-vous faire ?", buttons=[
            {"title": "Retour", "payload": "/go_back"},
            {"title": "Obtenir de l'aide", "payload": "/ask_help"}
        ])

        return []
